In Home.jsx, replace the current hero section entirely with a sticky scroll-driven experience. The 3D model is built from scratch using Three.js — no GLB file, no Spline embed, pure code.

STEP 1 — Install dependencies if not already installed:
bashnpm install three @react-three/fiber @react-three/drei framer-motion

STEP 2 — Hero section HTML/JSX structure:
jsximport { useRef, useEffect, useState } from 'react'
import { useScroll, useTransform, motion, AnimatePresence } from 'framer-motion'
import * as THREE from 'three'

export default function Home() {
  const heroWrapperRef = useRef(null)
  const canvasRef = useRef(null)
  const sceneRef = useRef({})

  const { scrollYProgress } = useScroll({
    target: heroWrapperRef,
    offset: ['start start', 'end end']
  })

  return (
    <motion.div>

      {/* STICKY HERO WRAPPER — 3 screens tall */}
      <div ref={heroWrapperRef} style={{ height: '300vh', position: 'relative' }}>

        {/* STICKY CONTENT — stays pinned while scrolling */}
        <div style={{
          position: 'sticky',
          top: 0,
          height: '100vh',
          display: 'flex',
          alignItems: 'center',
          overflow: 'hidden',
          background: '#0a0a0a'
        }}>

          {/* LEFT SIDE — text that changes per act */}
          <div style={{ flex: 1, padding: '0 4rem', zIndex: 10 }}>
            <ActText scrollProgress={scrollYProgress} />
          </div>

          {/* RIGHT SIDE — Three.js canvas */}
          <div style={{ flex: 1, height: '100vh', position: 'relative' }}>
            <canvas
              ref={canvasRef}
              style={{ width: '100%', height: '100%', display: 'block' }}
            />
          </div>

        </div>
      </div>

      {/* REST OF PAGE — appears after sticky section ends */}
      <StatsCounter />
      <HowItWorks />
    </motion.div>
  )
}

    const len = dir.length()
    const mid = new THREE.Vector3((x1+x2)/2,(y1+y2)/2,(z1+z2)/2)
    const g = new THREE.CylinderGeometry(r,r,len,8)
    const m = new THREE.Mesh(g, mat(col,0.3,0.7))
    m.position.copy(mid)
    m.quaternion.setFromUnitVectors(new THREE.Vector3(0,1,0), dir.normalize())
    m.castShadow = true
    return m

    mouseY = -(e.clientY / window.innerHeight - 0.5) * 2
  }
  window.addEventListener('mousemove', onMouseMove)

  // ---- LERP HELPER ----
  const lerp = (a, b, t) => a + (b - a) * t

  // ---- ANIMATION LOOP ----
  let time = 0
  let animId
  let keypointOpacities = new Array(keypointSpheres.length).fill(0)

  function animate() {
    animId = requestAnimationFrame(animate)
    time += 0.012

    // Mouse follow
    group.rotation.y += (mouseX * 0.5 - group.rotation.y) * 0.05
    group.rotation.x += (mouseY * 0.08 - group.rotation.x) * 0.05

    // Breathing
    group.position.y = Math.sin(time * 0.8) * 0.02

    // Ball animation
    ballGroup.position.y = 1.88 + Math.abs(Math.sin(time * 1.4)) * 0.08
    ballGroup.rotation.y = time * 1.5

    // Rim light pulse
    rimLight.intensity = 2.5 + Math.sin(time * 2) * 0.5

    // ---- ACT TRANSITIONS ----
    const s = currentScroll

    // ACT 1 → ACT 2 transition (scroll 0.15 to 0.40)
    const act2Progress = Math.max(0, Math.min(1, (s - 0.15) / 0.25))

    // ACT 2 → ACT 3 transition (scroll 0.50 to 0.75)
    const act3Progress = Math.max(0, Math.min(1, (s - 0.50) / 0.25))

    // Keypoint spheres appear staggered in Act 2
    keypointSpheres.forEach((sphere, i) => {
      const delay = i / keypointSpheres.length
      const targetOpacity = act2Progress > delay ? Math.min(1, (act2Progress - delay) * 5) : 0
      keypointOpacities[i] = lerp(keypointOpacities[i], targetOpacity, 0.08)
      sphere.material.opacity = keypointOpacities[i]
      // Pulse in Act 2
      sphere.scale.setScalar(1 + Math.sin(time * 2 + i) * 0.08 * act2Progress)
    })

    // Skeleton lines appear with Act 2
    skeletonLines.forEach((line, i) => {
      const delay = i / skeletonLines.length * 0.5
      const targetOpacity = act2Progress > delay ? Math.min(0.7, (act2Progress - delay) * 3) : 0
      line.material.opacity = lerp(line.material.opacity, targetOpacity, 0.05)
    })

    // Athlete body fades OUT in Act 3
    athleteMeshes.forEach(mesh => {
      const targetOpacity = 1 - act3Progress
      mesh.material.opacity = lerp(mesh.material.opacity, targetOpacity, 0.05)
    })

    // Wheelchair wireframe fades IN in Act 3
    wireframeMesh.material.opacity = lerp(
      wireframeMesh.material.opacity,
      act3Progress * 0.6,
      0.05
    )

    // Keypoints get brighter in Act 3
    keypointSpheres.forEach(sphere => {
      sphere.material.emissiveIntensity = lerp(
        sphere.material.emissiveIntensity,
        0.8 + act3Progress * 2,
        0.05
      )
    })

    // Ball fades slightly in Act 3
    ballGroup.children.forEach(m => {
      if (m.material) {
        m.material.transparent = true
        m.material.opacity = lerp(m.material.opacity ?? 1, 1 - act3Progress * 0.7, 0.05)
      }
    })

    renderer.render(scene, camera)
  }

  animate()

  // Cleanup
  return () => {
    cancelAnimationFrame(animId)
    unsubscribe()
    window.removeEventListener('mousemove', onMouseMove)
    renderer.dispose()
  }

}, [])
